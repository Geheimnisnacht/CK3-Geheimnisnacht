namespace = wh_bretonnia_landed_knight_check

# Event for a newly capable character to become a knight of the lady and start the track
wh_bretonnia_landed_knight_check.001 = {
	type = character_event
	title = wh_bretonnia_landed_knight_check.t
	desc = wh_bretonnia_landed_knight_check.desc
	theme = realm

	left_portrait = {
		character = root
		animation = personality_honorable
	}

	trigger = {
		faith = {
			has_doctrine = tenet_knightly_virtues
		}
		NOT = { has_trait = knight_lady }
		NOT = { has_character_modifier = wh_bretonnia_unproven_noble }
		is_adult = yes
		OR = {
			culture = { has_cultural_pillar = martial_custom_equal }
			AND = {
				is_male = yes
				culture = { has_cultural_pillar = martial_custom_male_only }
			}
			AND = {
				is_female = yes
				culture = { has_cultural_pillar = martial_custom_female_only }
			}
		}
		OR = {
			NOT = { exists = scope:title }
			scope:title = {
				tier >= tier_county
				this = root.primary_title
			}
		}
	}

	option = {
		name = wh_bretonnia_landed_knight_check.001.a

		add_trait = knight_lady_errant
		if = {
			limit = {
				OR = {
					is_ai = no
					is_landed = yes
				}
			}
			trigger_event = {
				id = wh_bretonnia_errantry.0002
				days = 1
			}
		}

		ai_chance = {
			base = 75
			modifier = {
				add = 25
				has_trait = brave
			}
			modifier = {
				add = 25
				has_trait = zealous
			}
			modifier = {
				add = -30
				has_trait = craven
			}
		}
	}

	option = {
		name = wh_bretonnia_landed_knight_check.001.b

		add_trait = knight_lady_realm
		add_character_modifier = {
			modifier = wh_bretonnia_unproven_noble
			years = 10
		}

		ai_chance = {
			base = 25
			modifier = {
				add = 30
				has_trait = craven
			}
			modifier = {
				add = -25
				has_trait = brave
			}
			modifier = {
				add = -25
				has_trait = zealous
			}
		}
	}

	after = {
		if = {
			limit = {
				OR = {
					is_ai = no
					is_landed = yes
				}
				NOT = {
					any_owned_story = { type = story_cycle_martial_lifestyle_warhorse }
				}
			}
			trigger_event = {
				id = wh_bretonnia_landed_knight_check.002
				days = { 3 7 }
			}
		}
	}
}

# Warhorse purchase event for the newly appointed knight of the lady
wh_bretonnia_landed_knight_check.002 = {
	type = character_event
	title = wh_bretonnia_landed_knight_check.002.t
	desc = wh_bretonnia_landed_knight_check.002.desc
	theme = martial_chivalry_focus
	override_background = {
		reference = courtyard
	}

	left_portrait = {
		character = root
		animation = standing_horse
		camera = camera_event_standing_with_horse_left
	}

	trigger = {
		has_trait = knight_lady
		NOT = {
			any_owned_story = { type = story_cycle_martial_lifestyle_warhorse }
		}
	}

	option = {
		name = wh_bretonnia_landed_knight_check.002.a
		trigger = {
			medium_gold_value_trigger = yes
		}
		show_as_unavailable = {
			always = yes
		}

		remove_short_term_gold = medium_gold_value
		start_warhorse_story_cycle_effect = yes

		ai_chance = {
			base = 75
			modifier = {
				add = 25
				has_trait = brave
			}
			modifier = {
				add = 25
				has_trait = arrogant
			}
			modifier = {
				add = -50
				has_trait = greedy
			}
		}
	}

	option = {
		name = wh_bretonnia_landed_knight_check.002.b

		ai_chance = {
			base = 25
			modifier = {
				add = 50
				has_trait = greedy
			}
		}
	}
}
