# Event for the new knight errant to set out on their errantry tour
wh_bretonnia_errantry.0001 = {
	type = character_event
	title = wh_bretonnia_errantry.0001.t
	desc = wh_bretonnia_errantry.0001.desc
	theme = travel

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		has_trait = knight_lady_errant
		NOT = { has_character_flag = wh_on_errantry_tour }
	}

	option = {
		name = wh_bretonnia_errantry.0001.a
		wh_bretonnia_errantry_tour_start_effect = yes
	}
}

# Hidden event to start the errantry tour once the knight errant is available, retrying every 30 days
wh_bretonnia_errantry.0002 = {
	type = character_event
	hidden = yes

	trigger = {
		has_trait = knight_lady_errant
		NOT = { has_character_flag = wh_on_errantry_tour }
	}

	immediate = {
		if = {
			limit = { is_available_adult = yes }
			trigger_event = wh_bretonnia_errantry.0001
		}
		else = {
			trigger_event = {
				id = wh_bretonnia_errantry.0002
				days = 30
			}
		}
	}
}
