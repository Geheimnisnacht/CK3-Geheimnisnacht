language_reikspiel = {
	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = language_reikspiel
		}
	}
	ai_will_do = {
		value = 10
		if = {
			limit = { has_cultural_pillar = language_reikspiel }
			multiply = 10
		}
	}
	
	color = wh_steel_grey
}
