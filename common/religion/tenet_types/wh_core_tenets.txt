	icon = tenet_assassin

		custom_description = {
			text = incompatible_tenet_pacifism_trigger
			NOT = { tenet:tenet_pacifism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = fedayeen_required_doctrines_trigger
			OR = {
				tenet:tenet_warmonger = { is_in_list = selected_doctrines }
				tenet:tenet_unrelenting_faith = { is_in_list = selected_doctrines }
				tenet:tenet_consolamentum = { is_in_list = selected_doctrines }
				tenet:tenet_armed_pilgrimages = { is_in_list = selected_doctrines }

			}
		}
	}

	piety_cost = {
		if = {
			limit = {
				OR = {
					rite_has_tenet = tenet_warmonger

		}

	parameters = {

	}

	character_modifier = {
		controlled_province_advantage = 5
