below_settlement_limit = {
	trigger_if = {
		# limit = { NOT = { dynasty ?= { has_dynasty_modifier = easy_arnor_settle } } }
		trigger_if = {
			limit = { highest_held_title_tier = tier_county }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 1
			}
		}
		trigger_else_if = {
			limit = { highest_held_title_tier = tier_duchy }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 2
			}
		}
		trigger_else_if = {
			limit = { highest_held_title_tier = tier_kingdom }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 3
			}
		}
		trigger_else = {
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 4
			}
		}
	}
	trigger_else = {
		trigger_if = {
			limit = { highest_held_title_tier = tier_county }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 3
			}
		}
		trigger_else_if = {
			limit = { highest_held_title_tier = tier_duchy }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 4
			}
		}
		trigger_else_if = {
			limit = { highest_held_title_tier = tier_kingdom }
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 5
			}
		}
		trigger_else = {
			any_held_title = {
				tier = tier_county
				title_province ?= { has_holding_type = settlement_holding }
				count < 6
			}
		}
	}
}

lotr_grant_settlement_away_with_ambitious_trait_trigger = {
	trigger_if = {
		limit = {
			any_in_list = {
				list = target_titles
				tier <= tier_county
				title_province = { has_holding_type = settlement_holding }
			}
		}
		above_settlement_limit = no
	}

lotr_WAC_basic_building_trigger = {
	trigger_if = {
		limit = { var:title_tier_to_colonise != flag:county }
		var:title_to_colonise = {
			any_de_jure_county = {
				title_province = {
					has_holding_type = settlement_holding
					NOT = { has_building_with_flag = obstacle }
					has_free_building_slot = yes
				}
			}
		}
	}
	trigger_else = {
		var:title_to_colonise = {
			title_province = {
				has_holding_type = settlement_holding
				NOT = { has_building_with_flag = obstacle }
				has_free_building_slot = yes
			}
		}
	}
}

lotr_WAC_basic_holding_trigger = {
	trigger_if = {
		limit = { var:title_tier_to_colonise != flag:county }
		var:title_to_colonise = {
			any_de_jure_county = {
				title_province = {
					has_holding_type = settlement_holding
					NOT = { has_building_with_flag = obstacle }
					has_building = settlement_$HOLDING_LEVEL$
				}
			}
		}
	}
	trigger_else = {
		var:title_to_colonise = {
			title_province = {
				has_holding_type = settlement_holding
				NOT = { has_building_with_flag = obstacle }
				has_building = settlement_$HOLDING_LEVEL$
			}
		}
	}
}
